# LumeOS boot configuration for the Raspberry Pi Zero W (BCM2835, ARM1176JZF-S).
#
# This file is copied into the boot partition of the SD card image built by
# `make image`.  The Raspberry Pi boot firmware (start.elf) reads it before it
# loads the kernel; the settings below are the ones LumeOS actually depends on.
#
# Reference: Raspberry Pi documentation, config.txt ("legacy" options used by
# the 32-bit boot firmware).

# Load our kernel from the boot partition.  The firmware's default file name is
# kernel.img too, but naming it explicitly keeps the intent obvious.
kernel=kernel.img

# Physical address the kernel is loaded at and jumped to.  0x8000 is the
# 32-bit default and is exactly the load address LumeOS' linker script uses
# (kernel/ld/lumeos.ld: LMA 0x8000, VMA 0xC0008000).
kernel_address=0x8000

# Enable the serial port.  On a board with onboard Bluetooth (Pi Zero W, Pi 3,
# ...) the *primary* UART that this enables is the mini UART, not the PL011, so
# the next option matters as well.
enable_uart=1

# LumeOS drives the PL011 (UART0, base 0x20201000).  On a Pi Zero W the PL011 is
# wired to the Bluetooth modem by default, so it has to be handed back to
# GPIO14/15 for the kernel's console to be visible on the header.  Without this
# line a perfectly working kernel prints into the Bluetooth chip.
# See docs/hardware.md and docs/boot-pi.md.
dtoverlay=disable-bt

# LumeOS does not read ATAGS or a command line from the firmware: memory sizes
# come from the mailbox property interface (GET_ARM_MEMORY) and the kernel
# command line is empty.  This stops start.elf from writing an ATAGS block into
# low memory.
disable_commandline_tags=1

# Uncomment to make the firmware itself print early boot progress on the serial
# port.  Useful when the board does not appear to boot at all; not needed for
# normal operation.
#uart_2ndstage=1

# LumeOS brings up its own framebuffer request through the mailbox property
# interface when the framebuffer driver is enabled (see docs/roadmap.md), so no
# hdmi_* settings are needed here yet.
